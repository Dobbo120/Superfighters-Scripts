/*

*/

public static void OnStartup(){
	
	//Add Applicable Weapons
	foreach(ProjectileItem item in Enum.GetValues(typeof(ProjectileItem))){
		
		if(item != ProjectileItem.NONE)
			ProjectileItems.Add(item);
	}
	
	//Set File
	ConfigFile = Game.GetSharedStorage("BDCONFIG");
	
	//Initialize Events
	Updater = Events.UpdateCallback.Start(OnUpdate);
	ProjectileHit = Events.ProjectileHitCallback.Start(OnProjectileHit);
	ProjectileCreated = Events.ProjectileCreatedCallback.Start(OnProjectileCreated);	
	ObjectCreated = Events.ObjectCreatedCallback.Start(OnObjectCreated);	
	
	m_presentdayTerminated = Events.ObjectTerminatedCallback.Start(presentdayTerminated);
	
	UserMessage = Events.UserMessageCallback.Start(OnUserMessage);
	
	//Auto Randomize Items
	if(GetConfigVar<bool>("ITEMS_AUTO_RANDOMIZE", false)){
		WeaponsRandomize();
	}
	
	//Add Players To Tracking List
	foreach(IPlayer ply in Game.GetPlayers()){
		PlayerList.Add(new gPlayer(ply));
	}
}

public void OnShutdown(){
	
	if(Updater != null)
		Updater.Stop();	
	
	if(ProjectileHit != null)
		ProjectileHit.Stop();	
	
	if(ProjectileCreated != null)
		ProjectileCreated.Stop();	
	
	if(ObjectCreated != null)
		ObjectCreated.Stop();
	
	if(UserMessage != null)
		UserMessage.Stop();
}

/*
	User Commands and Config
*/
public static void OnUserMessage(UserMessageCallbackArgs args){
	
	if(args.IsCommand && args.User.IsModerator){
		string[] c_args = args.CommandArguments.Split(' ');
		
		switch(args.Command){
			
			//List Commands
			case "BD_HELP": {
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage("BULLETS DELUXE", Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				//Ammo Commands
				Game.ShowChatMessage("'/BD_AMMOS_GET_ALL'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Get all bullet types", colDesc, args.User.UserIdentifier);
				
				Game.ShowChatMessage("'/BD_AMMOS_GET_ENABLED'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Get all enabled bullet types", colDesc, args.User.UserIdentifier);
				
				Game.ShowChatMessage("'/BD_AMMOS_SET [X] [0 or 1]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Set bullet type[X] enabled or disabled [Y]", colDesc, args.User.UserIdentifier);
				
				Game.ShowChatMessage("'/BD_AMMOS_RESET'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Reset all disabled bullet types", colDesc, args.User.UserIdentifier);
					
				//Item Commands
				Game.ShowChatMessage("'/BD_ITEMS_GET_ALL'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Get all item's set bullet type", colDesc, args.User.UserIdentifier);				
					
				Game.ShowChatMessage("'/BD_ITEMS_GET [X]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Get item[X]'s set bullet type", colDesc, args.User.UserIdentifier);

				Game.ShowChatMessage("'/BD_ITEMS_SET [X] [Y]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Set item[X] to bullet type [Y]", colDesc, args.User.UserIdentifier);				
					
				Game.ShowChatMessage("'/BD_ITEMS_SET_ALL [X]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Sets all items to bullet type [X]", colDesc, args.User.UserIdentifier);
				
				Game.ShowChatMessage("'/BD_ITEMS_RANDOMIZE'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Randomize all items with enabled bullet types", colDesc, args.User.UserIdentifier);					
					
				Game.ShowChatMessage("'/BD_ITEMS_AUTORANDOMIZE [0 or 1]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Automatically randomize the items every round", colDesc, args.User.UserIdentifier);				
					
				Game.ShowChatMessage("'/BD_ITEMS_RESET'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Reset all set bullet type items", colDesc, args.User.UserIdentifier);
					
				Game.ShowChatMessage("'/BD_CRATES_SET'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Set drop rate of crate", colDesc, args.User.UserIdentifier);				
					
				Game.ShowChatMessage("'/BD_GIVE [X] [Y]'", colHeader, args.User.UserIdentifier);
					Game.ShowChatMessage("  Give player[X] bullet type [Y]", colDesc, args.User.UserIdentifier);
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage("SCROLL UP FOR MORE COMMANDS", Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				break;
			}			
			
			//List All Bullet Types
			case "BD_AMMOS_GET_ALL": {
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage("ALL BULLET TYPES", Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				for(int i = 0;i < AmmoTypes.Length;i ++){
					Game.ShowChatMessage(i + ". " + AmmoTypes[i].Name, Color.White, args.User.UserIdentifier);
				}
				Game.ShowChatMessage(AmmoTypes.Length + ". None", Color.White, args.User.UserIdentifier);
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				break;
			}

			//List Enabled Bullet Types
			case "BD_AMMOS_GET_ENABLED": {
				string[] Disabled = GetConfigVar<string[]>("DISABLED_AMMOS", new string[] {""});
			
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage("ENABLED BULLET TYPES", Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
			
				for(int i = 0;i < AmmoTypes.Length;i ++){
					
					if(!Disabled.Contains(AmmoTypes[i].Name)){
						Game.ShowChatMessage(i + ". " + AmmoTypes[i].Name, Color.White, args.User.UserIdentifier);
					}else{
						Game.ShowChatMessage(i + ". " + AmmoTypes[i].Name, new Color(50, 50, 50), args.User.UserIdentifier);
					}
				}
				Game.ShowChatMessage(AmmoTypes.Length + ". None", Color.White, args.User.UserIdentifier);
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				break;
			}			
			
			//Disable Or Enable Bullet Bullet Types
			case "BD_AMMOS_SET": {
				
				if(c_args.Length == 2){
					string[] Disabled = GetConfigVar<string[]>("DISABLED_AMMOS", new string[] {""});
					
					int BulletID = GetAmmo(c_args[0]);
					
					if(BulletID >= 0 && BulletID < AmmoTypes.Length){
						List<string> AmmoList = Disabled.ToList();
						int index = AmmoList.FindIndex(x => x == AmmoTypes[BulletID].Name);
						
						//Enable
						if(c_args[1] == "1" || c_args[1].ToUpper() == "ENABLE"){
							
							if(index >= 0){
								Game.ShowChatMessage("[" + AmmoTypes[BulletID].Name + "] enabled", colPassed, args.User.UserIdentifier);
								AmmoList.RemoveAt(index);
							}else{
								Game.ShowChatMessage("[" + AmmoTypes[BulletID].Name + "] already enabled", colPassed, args.User.UserIdentifier);
							}
							
						//Disable
						}else if(c_args[1] == "0" || c_args[1].ToUpper() == "DISABLE"){
							
							if(index >= 0){
								Game.ShowChatMessage("[" + AmmoTypes[BulletID].Name + "] already disabled", colPassed, args.User.UserIdentifier);								
							}else{
								Game.ShowChatMessage("[" + AmmoTypes[BulletID].Name + "] disabled", colPassed, args.User.UserIdentifier);
								AmmoList.Add(AmmoTypes[BulletID].Name);
							}

						//Failed To Find Enable or Disabled
						}else{
							Game.ShowChatMessage("/BD_AMMOS_SET : Y must equal 0, 1, enable, or disable", colFailed, args.User.UserIdentifier);	
						}							
						SetConfigVar<string[]>("DISABLED_AMMOS", AmmoList.ToArray());
					
					//Failed Out Of Bounds
					}else{
						Game.ShowChatMessage("/BD_AMMOS_SET : X out of bounds of array", colFailed, args.User.UserIdentifier);			
					}
				
				//Failed To Find Args
				}else{
					Game.ShowChatMessage("/BD_AMMOS_SET [BulletType] [0, 1]", colFailed, args.User.UserIdentifier);		
				}
				break;
			}
			
			//Reset All Disabled Ammos
			case "BD_AMMOS_RESET": {			
				SetConfigVar<string[]>("DISABLED_AMMOS", new string[]{""});
				Game.ShowChatMessage("Enabled all ammo types", colPassed, args.User.UserIdentifier);		
				break;
			}			
			
			case "BD_ITEMS_RANDOMIZE": {
				WeaponsRandomize();
				Game.ShowChatMessage("Randomized all items", colPassed, args.User.UserIdentifier);	
				break;
			}			
			
			case "BD_ITEMS_AUTORANDOMIZE": {

				if(c_args.Length == 1 && c_args[0] != ""){
					
					if(c_args[0] == "0" || c_args[0].ToUpper() == "DISABLE"){
						SetConfigVar<bool>("ITEMS_AUTO_RANDOMIZE", false);
						Game.ShowChatMessage("Disabled automatic randomization of items", colPassed, args.User.UserIdentifier);	
					}
					
					if(c_args[0] == "1" || c_args[0].ToUpper() == "ENABLE"){
						SetConfigVar<bool>("ITEMS_AUTO_RANDOMIZE", true);
						Game.ShowChatMessage("Enabled automatic randomization of items", colPassed, args.User.UserIdentifier);	
					}
				}else{
					Game.ShowChatMessage("/BD_ITEMS_AUTORANDOMIZE [0, 1]", colFailed, args.User.UserIdentifier);
				}
				break;
			}
			
			//Assign Bullet Type To Item
			case "BD_ITEMS_SET": {
			
				if(c_args.Length == 2){
					int WeaponID = GetWeapon(c_args[0]);
					int BulletID = GetAmmo(c_args[1]);
					
					//Not Default Values Set In Config
					if(WeaponID >= 1 && BulletID >= 0 && BulletID <= AmmoTypes.Length){
						
						//Not NONE Ammo
						if(BulletID != AmmoTypes.Length){
							SetConfigVar<string>(GetWeaponName(WeaponID) + "_AMMO", AmmoTypes[BulletID].Name);
							Game.ShowChatMessage("[" + GetWeaponName(WeaponID) + "] ammo set to [" + AmmoTypes[BulletID].Name + "]", colPassed, args.User.UserIdentifier);
							
						//NONE Ammo
						}else{
							SetConfigVar<string>(GetWeaponName(WeaponID) + "_AMMO", "None");
							Game.ShowChatMessage("[" + GetWeaponName(WeaponID) + "] ammo set to [None]", colPassed, args.User.UserIdentifier);
						}
						
					//Failed 
					}else if(WeaponID < 1){
						Game.ShowChatMessage("/BD_ITEMS_SET : [WeaponItem] invalid weapon name or index", colFailed, args.User.UserIdentifier);	
						
					}else if(!((BulletID >= 0 && BulletID < AmmoTypes.Length) || BulletID == AmmoTypes.Length)){
						Game.ShowChatMessage("/BD_ITEMS_SET : [BulletType] invalid bullet name or index", colFailed, args.User.UserIdentifier);	
					}
					
				//Failed To Find Args
				}else{
					Game.ShowChatMessage("/BD_ITEMS_SET [WeaponItem] [BulletType]", colFailed, args.User.UserIdentifier);		
				}		

				//Check for crate enable
				CrateCheck();		
				
				break;
			}				
			
			//Assign Bullet Type To All Items
			case "BD_ITEMS_SET_ALL": {
			
				if(c_args.Length == 1){		
					int BulletID = GetAmmo(c_args[0]);
					
					//Not Default Values Set In Config
					if(BulletID >= 0 && BulletID <= AmmoTypes.Length){
						
						//Not NONE Ammo
						if(BulletID != AmmoTypes.Length){
							
							foreach(ProjectileItem item in ProjectileItems){
								SetConfigVar<string>(GetWeaponName(item) + "_AMMO", AmmoTypes[BulletID].Name);	
							}
							Game.ShowChatMessage("All items ammo set to [" + AmmoTypes[BulletID].Name + "]", colPassed, args.User.UserIdentifier);
							
						//NONE Ammo
						}else{
							
							foreach(ProjectileItem item in ProjectileItems){
								SetConfigVar<string>(GetWeaponName(item) + "_AMMO", "None");	
							}
							Game.ShowChatMessage("All items ammo set to [None]", colPassed, args.User.UserIdentifier);
						}
						
					//Failed 
					}else if(!((BulletID >= 0 && BulletID < AmmoTypes.Length) || BulletID == AmmoTypes.Length)){
						Game.ShowChatMessage("/BD_ITEMS_SET_ALL : [BulletType] invalid bullet name or index", colFailed, args.User.UserIdentifier);	
					}
					
				//Failed To Find Args
				}else{
					Game.ShowChatMessage("/BD_ITEMS_SET_ALL [BulletType]", colFailed, args.User.UserIdentifier);		
				}	

				//check for crate enable
				CrateCheck();			
				
				break;
			}		
			
			//Reset Default Ammos
			case "BD_ITEMS_RESET": {
				SetConfigVar<bool>("SPECIAL_CRATE_ENABLED", true);
				
				//Cycle Each WeaponItem
				foreach(ProjectileItem item in ProjectileItems){	
					SetConfigVar<string>(GetWeaponName(item) + "_AMMO", "None");					
				}
				Game.ShowChatMessage("All weapons ammo set to [None]", colPassed, args.User.UserIdentifier);		
				break;
			}			
			
			//Get Ammo For Specific WeaponItem
			case "BD_ITEMS_GET": {
				
				if(c_args.Length == 1 && c_args[0] != ""){
					int WeaponID = GetWeapon(c_args[0]);
					
					if(WeaponID != -1){
						Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
						Game.ShowChatMessage("WEAPONITEM BULLETTYPE", Color.White, args.User.UserIdentifier);
						Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
						
						string AmmoName = GetConfigVar<string>(GetWeaponName(WeaponID) + "_AMMO", "None");
						Game.ShowChatMessage("[" + GetWeaponName(WeaponID) + "] ammo is [" + AmmoName + "]", Color.White, args.User.UserIdentifier);
						
						Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
					}else{
						Game.ShowChatMessage("/BD_ITEMS_GET : WeaponItem invalid weapon name or index",  Color.White, args.User.UserIdentifier);
					}
					
				}else{
					Game.ShowChatMessage("/BD_ITEMS_GET [WeaponItem]", colFailed, args.User.UserIdentifier);		
				}
				break;
			}
			
			//Get Ammo For All WeaponItem
			case "BD_ITEMS_GET_ALL": {
				bool found = false;
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage("ALL WEAPONITEM BULLETTYPES", Color.White, args.User.UserIdentifier);
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);
				
				foreach(ProjectileItem item in ProjectileItems){					
					string AmmoName = GetConfigVar<string>(GetWeaponName(item) + "_AMMO", "None");
					
					//Weapon Item Not None
					if(AmmoName != "None"){
						found = true;
						Game.ShowChatMessage("[" + GetWeaponName(item) + "] ammo is [" + AmmoName + "]", Color.White, args.User.UserIdentifier);
					}
				}
			
				if(!found)
					Game.ShowChatMessage("No items set to ammo types", Color.White, args.User.UserIdentifier);
				
				Game.ShowChatMessage(cmdSeperator, Color.White, args.User.UserIdentifier);				
				break;
			}
			
			case "BD_CRATES_SET": {
				
				if(c_args.Length == 1 && c_args[0] != ""){
					int rate = 0;
					
					if(Int32.TryParse(c_args[0], out rate)){
						
						if(0 <= rate && rate <= 100){
							SetConfigVar<int>("SPECIAL_CRATE_DROP_RATE", rate);
							Game.ShowChatMessage("Special crate drop chance set to " + rate + "%", colPassed, args.User.UserIdentifier);
						}else{
							Game.ShowChatMessage("/BD_CRATES_SET : Drop chance out of range, 0 - 100", colFailed, args.User.UserIdentifier);
						}
					}else{
						Game.ShowChatMessage("/BD_CRATES_SET : Drop non integer type", colFailed, args.User.UserIdentifier);
					}
				}else{
					Game.ShowChatMessage("/BD_CRATES_SET [DropChance 0% - 100%]", colFailed, args.User.UserIdentifier);
				}
				
				break;
			}
			
			case "BD_GIVE": {
				
				if(c_args.Length == 2){
					int PlayerID = GetPlayer(c_args[0]);
					int BulletID = GetAmmo(c_args[1]);				
				
					if(PlayerID != -1 && BulletID >= 0 && BulletID < AmmoTypes.Length){
						IPlayer target = Game.GetPlayer(PlayerID);
					
						if(target != null){
							PickupCustomBullet(target, AmmoTypes[BulletID]);
							Game.ShowChatMessage("Sent [" + AmmoTypes[BulletID].Name + "] to " + target.Name, colPassed, args.User.UserIdentifier);
						}
					}
					
				}else{
					Game.ShowChatMessage("/BD_GIVE [PlayerSlot or PlayerName] [BulletType]", colFailed, args.User.UserIdentifier);
				}
				break;
			}
		}
	}
}

public static void WeaponsRandomize(){
	//Disable Crates
	SetConfigVar<bool>("SPECIAL_CRATE_ENABLED", false);
	
	//Cycle Weapons
	foreach(ProjectileItem item in ProjectileItems){

public static void OnProjectileCreated(IProjectile[] projectiles){
	
	//Add Projectiles To Tracking
	foreach(IProjectile pp in projectiles){

		//Find If Default Type Set
		string Equip = GetConfigVar<string>(GetWeaponName(pp.ProjectileItem) + "_AMMO", "None");
		
		//Give New Properties
		pp.PowerupBounceActive = false;
		
		//Created Bullet Object
		Bullet bul = null;
		
		//Get Ammo From Player
		if(Equip == "None"){

			//Find Shooter
			gPlayer Source = GetgPlayer(Game.GetPlayer(pp.InitialOwnerPlayerID));
			
			//Shot By Player
			if(Source != null){
				int WeaponType = (int)pp.ProjectileItem;
				
				//From Primary
				if(WeaponType == (int)Source.Item1.WeaponItem && Source.Primary != null){
					bul = new Bullet(pp, Source.Primary);

				//From Secondary
				}else if(WeaponType == (int)Source.Item2.WeaponItem && Source.Secondary != null){
					bul = new Bullet(pp, Source.Secondary);
				}
			}
			
		//Get Ammo From Default Type, Owner Must Still Be a Player
		}else if(Equip != "None"){
			
			if(Game.GetPlayer(pp.InitialOwnerPlayerID) != null)
				bul = new Bullet(pp, GetAmmoByName(Equip));
		}
		
		if(bul != null){
			bul.Update();
			BulletList.Add(bul);
		}
	}
}

public static void CreateAmmoCrate(Vector2 pos){
	//Crates CustomID
	string cID = "DISABLED" + Game.TotalElapsedGameTime;
	
	//Joint To Hold Custom Crate Together
	IObjectWeldJoint Welder = (IObjectWeldJoint)Game.CreateObject("WeldJoint", pos);
	
	//Invisible Block For Game World Collision
	IObject Grid = Game.CreateObject("InvisibleBlockNoCollision", pos + new Vector2(-4, 4));
	Grid.SetSizeFactor(new Point(2, 2));
	
	CollisionFilter col = new CollisionFilter();
	col.CategoryBits = 32;
	col.MaskBits = 3;
	col.AboveBits = 0;
	col.BlockExplosions = false;
	col.BlockFire = false;
	col.BlockMelee = false;
	col.ProjectileHit = false;
	Grid.SetCollisionFilter(col);
	
	//Background Crate Image
	IObject BG = Game.CreateObject("BgRadar00A", pos, MathHelper.PI);

	//Disable Collision Tile With Crate Objects
	IObjectAlterCollisionTile Alter = (IObjectAlterCollisionTile)Game.CreateObject("AlterCollisionTile", pos);
	Alter.SetDisablePlayerMelee(true);
	Alter.SetDisableProjectileHit(true);
	Alter.SetDisabledCategoryBits(0xFFFF);
	Alter.SetDisabledMaskBits(0xFFFF);
	Alter.SetDisabledAboveBits(0xFFFF);

	//Crate Formation using Turret Crate Pieces
	for(int i = 0;i < 4;i ++){
		Vector2 Shift = Vector2.Zero;
		Shift.X = (float)Math.Cos((MathHelper.PI / 2) * i + MathHelper.PI / 4) * 5.65f;
		Shift.Y = (float)Math.Sin((MathHelper.PI / 2) * i + MathHelper.PI / 4) * 5.65f;
		
		IObject Box = Game.CreateObject("StreetsweeperCratePart", pos + Shift, MathHelper.PI / 2 * i - MathHelper.PI / 2);
		Box.SetBodyType(BodyType.Static);
		Box.CustomID = cID;	
		
		Welder.AddTargetObject(Box);
		Alter.AddTargetObject(Box);
	}
	
	//Button Players Interact With To Recieve Custom Ammunition
	IObjectActivateTrigger Button = (IObjectActivateTrigger)Game.CreateObject("ActivateTrigger", pos);
	Button.SetHighlightObject(Grid);
	Button.SetScriptMethod("PickupCustomBullet");

	//Add Crate Pieces To Main Weld
	Welder.AddTargetObject(Button);			
	Welder.AddTargetObject(Grid);
	Welder.AddTargetObject(BG);
	Welder.AddTargetObject(Alter);
	
	//Set CustomID's of Crate Pieces
	Button.CustomID = cID;
	Grid.CustomID = cID;
	BG.CustomID = cID;
	Welder.CustomID = cID;
	Alter.CustomID = cID;
	
	//Wait a few Secs for Game To Load In Objects Correctly
	CreateTimer(500, 1, "ValidateAmmoCrate", cID);	
}

public static void ValidateAmmoCrate(TriggerArgs args){

	//Only ObjectTriggers Will Use This Function
	if(args.Caller is IObject){
		string cID = (args.Caller as IObject).CustomID;
		
		//Destroy Unused Timer
		(args.Caller as IObject).Remove();
		
		//Set Each Create Object As Dynamic
		foreach(IObject obj in Game.GetObjectsByCustomID(cID)){
			obj.SetBodyType(BodyType.Dynamic);
		}
	}
}

public static bool PickupCustomBullet(IPlayer Player, Ammo Equip, bool ExtraAmmo = false){
	bool GotPickuped = true;
	
	//Find Currently Equiped Item
	RifleWeaponItem rifle = Player.CurrentPrimaryWeapon;
	HandgunWeaponItem handgun = Player.CurrentSecondaryWeapon;
	
	//Apply To Handgun
	if(Player.CurrentWeaponDrawn == WeaponItemType.Handgun && GetConfigVar<string>(GetWeaponName(handgun.WeaponItem) + "_AMMO", "None") == "None"){
		
		//Validate As Proper Player
		gPlayer ply = GetgPlayer(Player);

		if(ply != null)
			ply.GiveSecondaryAmmo(Equip);
		
	//Apply To Rifle
	}else if(Player.CurrentWeaponDrawn == WeaponItemType.Rifle && GetConfigVar<string>(GetWeaponName(rifle.WeaponItem) + "_AMMO", "None") == "None"){	

		//Validate As Proper Player
		gPlayer ply = GetgPlayer(Player);

		if(ply != null)
			ply.GivePrimaryAmmo(Equip);
		
	}else if(ExtraAmmo){
		gPlayer ply = GetgPlayer(Player);
		
		if(ply != null){
			//No Items Applied To
			
			//Save Ammo For Later
			if(ply.SavedAmmo < 2)
				ply.SavedAmmo ++;
			
			//Tell Player It Functioned
			Game.PlayEffect("CFTXT", Player.GetWorldPosition(), "Special Ammo");
		}else{
			GotPickuped = false;
		}
	}

	return GotPickuped;
}

	Ammo Type;

	public Bullet(IProjectile pp, Ammo am){
		Projectile = pp;
		Type = am;

	public void Update(){
		
		//Step Callback
		if(Type.OnUpdate != null){
			Type.OnUpdate(Projectile);
		}	
	}

	public void Hit(ProjectileHitArgs args){
		
		//Hit Callback
		if(Type.OnHit != null){
			Type.OnHit(Projectile, args);
		}
	}
};

public class gPlayer{

	public IPlayer Player;
	
	//Custom Bullet Storage
	public Ammo Primary = null;
	public Ammo Secondary = null;
	public int SavedAmmo = 0;
	
	//Store Items For Tracking Dropped Items
	public RifleWeaponItem Item1;
	public HandgunWeaponItem Item2;
	
	public gPlayer(IPlayer ply){
		Player = ply;
	}
	
	public void Update(){

		/*
			Find new items picked up
			and track dropped items.
			Activated when items not same as stored
		*/
		if(Player.CurrentPrimaryWeapon.WeaponItem != Item1.WeaponItem){

			//Not None? Track New Dropped Item If Has Bullet Type and no Default Type
			if(Item1.WeaponItem != WeaponItem.NONE && Primary != null && GetConfigVar<string>(GetWeaponName(Item1.WeaponItem) + "_AMMO", "None") == "None"){
				DroppedItems.Add(new dItem(Item1, Primary));
			}
	
			//Early Set Item
			Item1 = Player.CurrentPrimaryWeapon;
			Primary = null;

			//Find If Picked Up Weapon Is Dropped Item With Bullet Type
			for(int i = 0;i < DroppedItems.Count;i ++){
				dItem item = DroppedItems[i];
				
				if(RifleSame(item.rifle, Item1)){
					Primary = item.Type;
					DroppedItems.RemoveAt(i);
					break;
				}
			}
			
			//Find Default Ammo Type
			string Equip = GetConfigVar<string>(GetWeaponName(Item1.WeaponItem) + "_AMMO", "None");
			if(Equip != "None"){
				Primary = GetAmmoByName(Equip);
			}
			
			if(Item1.WeaponItem != WeaponItem.NONE)
				BulletInfo(Primary);			

			//Add Custom Ammo If Have Any Saved Up Ammo
			if(Primary == null){
				
				if(Item1.WeaponItem != WeaponItem.NONE){
					// DobEdit && SavedAmmo > 0
					if(GivePrimaryAmmo(GetRandomAmmo()))
						SavedAmmo = 1;
				}
			}			
		}
		
		if(Player.CurrentSecondaryWeapon.WeaponItem != Item2.WeaponItem){

			//Not None? Track New Dropped Item If Has Bullet Type and no Default Type
			if(Item2.WeaponItem != WeaponItem.NONE && Secondary != null && GetConfigVar<string>(GetWeaponName(Item2.WeaponItem) + "_AMMO", "None") == "None"){
				DroppedItems.Add(new dItem(Item2, Secondary));
			}

			//Early Set Item
			Item2 = Player.CurrentSecondaryWeapon;
			Secondary = null;
			
			//Find If Picked Up Weapon Is Dropped Item With Bullet Type
			for(int i = 0;i < DroppedItems.Count;i ++){
				dItem item = DroppedItems[i];
				
				if(HandgunSame(item.handgun, Item2)){
					Secondary = item.Type;
					DroppedItems.RemoveAt(i);
					break;
				}
			}
			
			//Find Default Ammo Type
			string Equip = GetConfigVar<string>(GetWeaponName(Item2.WeaponItem) + "_AMMO", "None");
			if(Equip != "None"){
				Secondary = GetAmmoByName(Equip);
			}

			if(Item2.WeaponItem != WeaponItem.NONE)
				BulletInfo(Secondary);
			
			//Add Custom Ammo If Have Any Saved Up Ammo
			if(Secondary == null){
				
				if(Item2.WeaponItem != WeaponItem.NONE){
					
					if(GiveSecondaryAmmo(GetRandomAmmo()))
						SavedAmmo = 1;
				}
			}
		}
		
		/*
			When Holding No Items or Lacking Bouncy Ammo When No Forced Type Reset Bullet Types
		*/
		if(Item1.WeaponItem == WeaponItem.NONE || (Item1.PowerupBouncingRounds == 0 && GetConfigVar<string>(GetWeaponName(Item1.WeaponItem) + "_AMMO", "None") == "None")){
			Primary = null;
		}		
		
		if(Item2.WeaponItem == WeaponItem.NONE || (Item2.PowerupBouncingRounds == 0 && GetConfigVar<string>(GetWeaponName(Item2.WeaponItem) + "_AMMO", "None") == "None")){
			Secondary = null;
		}
		
		//Track Weapons
		Item1 = Player.CurrentPrimaryWeapon;
		Item2 = Player.CurrentSecondaryWeapon;
	}
	
	/*
		When Bouncy Ammo Held and Not Already Given Bullet Type and No Forced Type
	*/
	public bool GivePrimaryAmmo(Ammo equip){

		if(GetConfigVar<string>(GetWeaponName(Item1.WeaponItem) + "_AMMO", "None") == "None"){				
				
			//Find Common Ammo and Mag Size	and if currentammo is smaller		
			int ammo = (Item1.MagSize < Item1.WeaponMagCapacity) ? Item1.WeaponMagCapacity : Item1.MagSize;
			int mags = (Item1.SpareMags <= Math.Ceiling(Item1.MaxCarriedSpareMags / 4.0)) ? (int)Math.Ceiling(Item1.MaxCarriedSpareMags / 4.0) : Item1.SpareMags;
			Player.SetCurrentPrimaryWeaponAmmo(ammo, mags, ProjectilePowerup.Bouncing);						

			Primary = equip;
			Item1 = Player.CurrentPrimaryWeapon;
			
			//Show Player Relevant Info On BulletType
			Game.PlayEffect("CFTXT", Player.GetWorldPosition(), Primary.Name);
			BulletInfo(Primary);
			
			return true;
		}
		
		return false;
	}
	
	public bool GiveSecondaryAmmo(Ammo equip){
		
		if(GetConfigVar<string>(GetWeaponName(Item2.WeaponItem) + "_AMMO", "None") == "None"){
		
			//Find Common Ammo and Mag Size	and if currentammo is smaller		
			int ammo = (Item2.MagSize < Item2.WeaponMagCapacity) ? Item2.WeaponMagCapacity : Item2.MagSize;
			int mags = (Item2.SpareMags <= Math.Ceiling(Item2.MaxCarriedSpareMags / 4.0)) ? (int)Math.Ceiling(Item2.MaxCarriedSpareMags / 4.0) : Item2.SpareMags;
			Player.SetCurrentSecondaryWeaponAmmo(ammo, mags, ProjectilePowerup.Bouncing);		
			
			Secondary = equip;
			Item2 = Player.CurrentSecondaryWeapon;
			
			//Show Player Relevant Info On BulletType
			Game.PlayEffect("CFTXT", Player.GetWorldPosition(), Secondary.Name);
			BulletInfo(Secondary);
			
			return true;
		}
		
		return false;
	}
	
	public void BulletInfo(Ammo Ammunition){
		
		//Stop Bots In SinglePlayer From Spamming Chat
		if(!Player.IsBot && Ammunition != null){
			
			//Name
			Game.ShowChatMessage(Ammunition.Name + ":", Color.White, Player.GetUser().UserIdentifier);
			
			//Author
			Game.ShowChatMessage("(" + Ammunition.Author + ")", Player.GetUser().UserIdentifier);
			
			//Description
			Game.ShowChatMessage(Ammunition.Description, Color.Grey, Player.GetUser().UserIdentifier);
		}
	}
};

public static int GetAmmo(string c_args){
	int BulletID = -1;
	
	if(!Int32.TryParse(c_args, out BulletID)){
		BulletID = -1;
		
		if(c_args.ToUpper() != "NONE"){
			
			for(int i = 0;i < AmmoTypes.Length;i ++){
				Ammo am = AmmoTypes[i];								

				if(am.Name.ToUpper().Split(' ')[0] == c_args.ToUpper()){
					BulletID = i;
					break;
				}
			}
			
		}else{
			BulletID = AmmoTypes.Length;
		}
	}
	
	return BulletID;
}

public static int GetPlayer(string c_args){
	IUser ret = null;
	
	foreach(IUser user in Game.GetActiveUsers()){
		
		if(user.Name.ToUpper() == c_args.ToUpper()){
			ret = user;
			break;
		}
	}
	
	if(ret == null){
		int id = 0;
		
		if(Int32.TryParse(c_args, out id)){
			
			foreach(IUser user in Game.GetActiveUsers()){
				
				if(user.GameSlotIndex == id){
					ret = user;
					break;
				}
			}
		}
	}
	
	return (ret != null && ret.GetPlayer() != null) ? ret.GetPlayer().UniqueID : -1;
}

public static string GetWeaponName(int WeaponID){
	string Name = "NONE";

	if(WeaponID > 0 && WeaponID.ToString() != ((ProjectileItem)WeaponID).ToString()){
		
		switch((ProjectileItem)WeaponID){
			
			case ProjectileItem.SUB_MACHINEGUN:{
				Name = "TOMMYGUN";
				break;
			}
			
			default: {
				Name = ((ProjectileItem)WeaponID).ToString();
				break;
			}
		}
	}
	return Name.ToUpper();
}

public static int GetWeapon(string c_args){
	int WeaponID = -1;
	
	//Try Parse Else Find By String Name
	if(!Int32.TryParse(c_args, out WeaponID)){
		WeaponID = -1;
		
		foreach(ProjectileItem item in ProjectileItems){		
			
			if(c_args.ToUpper() == GetWeaponName((int)item)){
				WeaponID = (int)item;
				break;			
			}
		}
	}else{
		
		if(GetWeaponName(WeaponID) == "NONE"){
			WeaponID = -1;
		}
	}	
	
	return WeaponID;
}

public static T GetConfigVar<T>(string Var, object Default){
	Object ret = null;

	//Non Null
	if(ConfigFile != null){
		
		if(ConfigFile.ContainsKey(Var))
			ret = ConfigFile.GetItem(Var);
		
		//Non Found? Set To Default
		if(ret == null && Default != null){
			SetConfigVar<T>(Var, Default);
			return (T)Default;
		}
		
		//Config Not Open
	}else{
		return (T)Default;
	}
	
	//Return Found or Not Found Data
	return (T)ret;
}

public static void SetConfigVar<T>(string Var, object Value){

	//Non Null
	if(ConfigFile != null && Value != null){	

		if(Value is string)
			ConfigFile.SetItem(Var, (string)Value);		
		
		if(Value is string[])
			ConfigFile.SetItem(Var, (string[])Value);
		
		if(Value is int)
			ConfigFile.SetItem(Var, (int)Value);		
		
		if(Value is int[])
			ConfigFile.SetItem(Var, (int[])Value);		
				
		if(Value is bool)
			ConfigFile.SetItem(Var, (bool)Value);

		if(Value is bool[])
			ConfigFile.SetItem(Var, (bool[])Value);		
		
		if(Value is float)
			ConfigFile.SetItem(Var, (float)Value);		
		
		if(Value is float[])
			ConfigFile.SetItem(Var, (float[])Value);
	}
}

public static void chainOnHit(IProjectile pp, ProjectileHitArgs args){
	IPlayer src = Game.GetPlayer(pp.InitialOwnerPlayerID);	

	if(args.IsPlayer && src != null){
		IPlayer hit = Game.GetPlayer(args.HitObjectID);
		IPlayer Next = null;
		float dist = -1;		

		foreach(IPlayer ply in Game.GetPlayers()){
			
			if(!ply.IsDead && ply != src && ply != hit && (src.GetTeam() == PlayerTeam.Independent || ply.GetTeam() != src.GetTeam())){
				float temp = Vector2.Distance(ply.GetWorldPosition(), pp.Position);

				if(dist == -1 || temp < dist){
					dist = temp;			
					Next = ply;
				}
			}
		}
		
		if(Next != null){

/*
	Homing
	Danger Ross 
	*edit By Ebomb09
*/
public static void homingUpdate(IProjectile pp){
	IPlayer Source = Game.GetPlayer(pp.InitialOwnerPlayerID);
	
	if(Source != null){
		double Angle = Math.Atan2(pp.Direction.Y, pp.Direction.X);
		double Minimum = MathHelper.PI / 180 * 10 * Game.SlowmotionModifier;
		IPlayer Target = null;
		float Dist = -1;
		
		foreach(IPlayer ply in Game.GetPlayers()){
			float r_Distance = Vector2.Distance(Source.GetWorldPosition(), ply.GetWorldPosition());
			
			if(ply != Source && !ply.IsDead && (ply.GetTeam() != Source.GetTeam() || Source.GetTeam() == PlayerTeam.Independent) && ((r_Distance < Dist) || Dist == -1)){
				Dist = r_Distance;
				Target = ply;
			}
		}
		
		if(Target != null){
			double d_Angle = Math.Atan2(Target.GetWorldPosition().Y - pp.Position.Y, Target.GetWorldPosition().X - pp.Position.X);
			
			double a = Angle - d_Angle;
			double b = Angle - d_Angle + MathHelper.PI * 2;
			double c = Angle - d_Angle - MathHelper.PI * 2;

			double d = 0;
			if(Math.Abs(a) < Math.Abs(b) && Math.Abs(a) < Math.Abs(c)){
				d = a;
			}else if(Math.Abs(b) < Math.Abs(a) && Math.Abs(b) < Math.Abs(c)){
				d = b;
			}else{
				d = c;
			}
			
			if(d > 0){
				Angle -= Minimum;
			}else if(d < 0){
				Angle += Minimum;
			}

			pp.Direction = new Vector2((float)Math.Cos(Angle), (float)Math.Sin(Angle));
		}
	}
}

public static void hypnoticammoHit(IProjectile pp, ProjectileHitArgs args){
	IPlayer src = Game.GetPlayer(pp.InitialOwnerPlayerID);	
	IPlayer tar = Game.GetPlayer(args.HitObjectID);

	if(src != null && tar != null){
		int chance = rnd.Next(0, 100);
		
		if((src.GetTeam() != tar.GetTeam() || src.GetTeam() == PlayerTeam.Independent) && chance <= 25){
			PlayerTeam sel = (src.GetTeam() == PlayerTeam.Independent) ? GetEmptyTeam() : src.GetTeam();
			tar.SetTeam(sel);
			src.SetTeam(sel);
			
			IObjectOnPlayerDeathTrigger trig = (IObjectOnPlayerDeathTrigger)Game.CreateObject("OnPlayerDeathTrigger");
			trig.SetPlayerSource(tar);
			trig.SetScriptMethod("RemoveFromTeam");
		}
	}
}

public static void RemoveFromTeam(TriggerArgs args){
	
	if(args.Caller is IObjectOnPlayerDeathTrigger){
		IObjectOnPlayerDeathTrigger trig = args.Caller as IObjectOnPlayerDeathTrigger;
		IPlayer tar = trig.GetPlayerSource();
		
		if(tar != null){
			PlayerTeam Team = tar.GetTeam();			
			IPlayer[] TeamPlayers = Game.GetPlayers().Where(x => x.GetTeam() == Team && !x.IsDead).ToArray();
			
			if(TeamPlayers.Length == 1)
				TeamPlayers[0].SetTeam(PlayerTeam.Independent);
		}
		trig.Remove();
	}	
}

public static PlayerTeam GetEmptyTeam(){
	List<PlayerTeam> Teams = new List<PlayerTeam>(){
		PlayerTeam.Team1,
		PlayerTeam.Team2,
		PlayerTeam.Team3,
		PlayerTeam.Team4
	};
	
	foreach(IPlayer ply in Game.GetPlayers().Where(x => !x.IsDead)){
		
		if(Teams.Contains(ply.GetTeam())){
			Teams.Remove(ply.GetTeam());			
		}
	}
	
	if(Teams.Count >= 1){
		return Teams[rnd.Next(0, Teams.Count)];
	}
	return PlayerTeam.Independent;
}

	projectile.FlagForRemoval();

public static int WeaponBulletsNum = 0;

public static void weaponUpdate(IProjectile pp){

	//Divide Parts of Name
	string[] ProjectileName = pp.ProjectileItem.ToString().Split('_');
	
	//Form Valid Object Name
	string WeaponName = "Wpn";
	foreach(string name in ProjectileName){
		WeaponName += name[0] + name.Substring(1, name.Length - 1).ToLower();
	}

	//Exceptions
	switch(pp.ProjectileItem){
		
		case ProjectileItem.SHOTGUN:
			WeaponName = "WpnPumpShotgun";
		break;		
		
		case ProjectileItem.SNIPER:
			WeaponName = "WpnSniperRifle";
		break;		
		
		case ProjectileItem.ASSAULT:
			WeaponName = "WpnAssaultRifle";
		break;		
		
		case ProjectileItem.SUB_MACHINEGUN:
			WeaponName = "WpnTommygun";
		break;		
	}
	
	//Create Object Properties
	float angle = (float)Math.Atan2(pp.Velocity.Y, pp.Velocity.X);
	IPlayer src = Game.GetPlayer(pp.InitialOwnerPlayerID);
	Vector2 pos = pp.Position;
	
	IObject item = Game.CreateObject(WeaponName, pos, angle, pp.Direction * 15, (float)Math.Cos(angle) * -20);
	item.CustomID = "__WEAPON_AMMO_" + WeaponBulletsNum + "__";

	//Move Pos of Gun Until Not Colliding With Player
	if(src != null){
		
		while(src.GetAABB().Intersects(item.GetAABB()))
			item.SetWorldPosition(item.GetWorldPosition() + pp.Direction);		
	}
	CreateTimer(10, 1, "WeaponMissile", "__WEAPON_AMMO_" + WeaponBulletsNum + "__");
	
	WeaponBulletsNum ++;
	
	//Remove Initial Projectile
	pp.FlagForRemoval();
}

public static void WeaponMissile(TriggerArgs args){
	
	if(args.Caller is IObject){
		//Find Custom ID
		string cID = (args.Caller as IObject).CustomID;
		
		//Clear Old Timer
		(args.Caller as IObject).CustomID = "";
		(args.Caller as IObject).Remove();
		
		//Get Weapon Item
		IObject item = Game.GetSingleObjectByCustomID(cID);
	
		//Still Exists?
		if(item != null){
			
			//Timer Control of Object
			CreateTimer(10, 0, "CheckFireWeapon", cID);
			CreateTimer(250, 0, "ProceedFireWeapon", cID);
		}
	}
}

public static void CheckFireWeapon(TriggerArgs args){
	
	if(args.Caller is IObject){
		string cID = (args.Caller as IObject).CustomID;
		
		//Find Weapon Still Exists
		bool found = false;
		foreach(IObject obj in Game.GetObjectsByCustomID(cID)){

			if(obj is IObjectWeaponItem){			
				obj.TrackAsMissile(false);
				bool IsInPlayer = false;
				
				foreach(IPlayer ply in Game.GetPlayers())
				{

					if(!ply.IsDead && obj.GetAABB().Intersects(ply.GetAABB())){
						IsInPlayer = true;
						
						Game.PlaySound("MeleeHitBlunt", ply.GetWorldPosition());
						
						PlayerModifiers mods = ply.GetModifiers();						
						float health = mods.CurrentHealth;
						health -= 5;
						
						if(health < 0){
							ply.Kill();
						}else{
							mods.CurrentHealth = health;
						}
						
						ply.SetModifiers(mods);
					}
				}
				
				//Not Colliding With Player
				if(!IsInPlayer && ((Math.Abs(obj.GetLinearVelocity().X) > 1) || Math.Abs(obj.GetLinearVelocity().Y) > 1 || Math.Abs(obj.GetAngularVelocity()) > 1)){
					found = true;
				}
				break;
			}
		}
		
		//Delete All Objects Associated
		if(!found){
			
			foreach(IObject obj in Game.GetObjectsByCustomID(cID)){				
				obj.Destroy();
			}
		}
	}
}

public static void ProceedFireWeapon(TriggerArgs args){
	
	if(args.Caller is IObject){
		string cID = (args.Caller as IObject).CustomID;
		
		//Find Weapon Item Object
		foreach(IObject obj in Game.GetObjectsByCustomID(cID)){
			
			if(obj is IObjectWeaponItem){

				//Projectile Properties
				float angle = obj.GetAngle();

				Game.SpawnProjectile((ProjectileItem)wpn.WeaponItem, obj.GetWorldPosition(), Direction * obj.GetFaceDirection());
			}
		}
	}
}

public class StunnedPlayer
{
	public static readonly float StunnedTime = 3000f;
	private IPlayer m_player;

	public int UniqueID
	{
		get { return m_player.UniqueID; }
	}

	private bool isStunned;
	public bool IsStunned
	{
		get
		{
			return isStunned;
		}
		private set
		{
			isStunned = value;
			if (isStunned)
			{
				m_player.SetInputEnabled(false);
				m_player.AddCommand(new PlayerCommand(PlayerCommandType.DeathKneelInfinite));
			}
			else
			{
				m_player.AddCommand(new PlayerCommand(PlayerCommandType.StopDeathKneel));
				m_player.SetInputEnabled(true);
			}
		}
	}

	public StunnedPlayer(IPlayer player) : this(player, player.GetWorldPosition())
	{
	}

	public StunnedPlayer(IPlayer player, Vector2 hitPosition)
	{
		m_player = player;
		IsStunned = true;

		Game.PlayEffect(EffectName.Electric, hitPosition);
	}

	private float m_stunnedTimer = 0;
	private float m_stunnedEffectTimer = 0;
	public void Update(float dt)
	{
		if (IsStunned)
		{
			m_stunnedTimer += dt;
			m_stunnedEffectTimer += dt;

			if (m_stunnedEffectTimer >= 400)
			{
				var position = m_player.GetWorldPosition();
				position.X += rnd.Next(-10, 10);
				position.Y += rnd.Next(-10, 10);

				Game.PlayEffect(EffectName.Electric, position);
				m_stunnedEffectTimer = 0;
			}

			if (m_stunnedTimer >= StunnedTime)
			{
				IsStunned = false;
			}
		}
	}
}

public static Dictionary<int, IProjectile> m_customBullets = new Dictionary<int, IProjectile>();
public static Dictionary<int, StunnedPlayer> m_stunnedPlayers = new Dictionary<int, StunnedPlayer>();

public static void electricUpdate(IProjectile projectile)
{
	if(m_electricUpdateEvent == null)
		m_electricUpdateEvent = Events.UpdateCallback.Start(electricUpdateEvent);
	
	ToEMPBullet(projectile);
}

public static void electricHit(IProjectile projectile, ProjectileHitArgs args)
{
	
	if (m_customBullets.ContainsKey(projectile.InstanceID))
	{
		var rndNum = rnd.Next(0, 100);
		if (rndNum < 1)
		{
			ElectrocuteRange(args.HitPosition);
		}
		if (1 <= rndNum && rndNum < 21)
		{
			var obj = Game.GetObject(args.HitObjectID);
			Electrocute(args);
		}

		m_customBullets.Remove(projectile.InstanceID);
	}
}

public static Vector2? m_empBlastCenter = null;
public static readonly float EMPBlastRadius = 15f;
public static float m_stunnedRangeTimer = 0f;
public static void ElectrocuteRange(Vector2 position)
{
	foreach (var player in Game.GetPlayers())
	{
		if (IsTouchingCircle(player.GetAABB(), position, EMPBlastRadius) && CanBeStunned(player))
		{
			Game.PlayEffect(EffectName.CustomFloatText, player.GetWorldPosition(), "stunned");
			m_stunnedPlayers.Add(player.UniqueID, new StunnedPlayer(player));
		}
	}

	for (var i = 0; i < 360; i += 72) // Play electric effect 5 times in circle (360 / 5 = 72)
	{
		var degree = i;
		var radianAngle = degree * Math.PI / 180.0f;

		var direction = new Vector2()
		{
			X = (float)Math.Cos(radianAngle),
			Y = (float)Math.Sin(radianAngle),
		};

		Game.PlayEffect(EffectName.Electric, position + direction * EMPBlastRadius);
	}

	m_empBlastCenter = position;
}

public static void Electrocute(ProjectileHitArgs args)
{
	var position = args.HitPosition;

	if (args.IsPlayer)
	{
		var player = (IPlayer)Game.GetObject(args.HitObjectID);

		if (CanBeStunned(player))
		{
			Game.PlayEffect(EffectName.CustomFloatText, position, "stunned");
			m_stunnedPlayers.Add(player.UniqueID, new StunnedPlayer(player, position));
		}
	}
}

public static bool CanBeStunned(IPlayer player)
{
	return !player.IsRemoved && !player.IsDead && !m_stunnedPlayers.ContainsKey(player.UniqueID);
}

public static Events.UpdateCallback m_electricUpdateEvent = null;
public static void electricUpdateEvent(float dt)
{
	UpdatePlayers(dt);

	m_stunnedRangeTimer += dt;
	if (m_stunnedRangeTimer >= StunnedPlayer.StunnedTime)
	{
		m_stunnedRangeTimer = 0;
		m_empBlastCenter = null;
	}
	if (m_empBlastCenter != null)
		Game.DrawCircle((Vector2)m_empBlastCenter, EMPBlastRadius, Color.Red);
}

public static void UpdatePlayers(float dt)
{
	foreach (var player in m_stunnedPlayers.Values.ToList())
	{
		player.Update(dt);

		if (!player.IsStunned)
		{
			m_stunnedPlayers.Remove(player.UniqueID);
		}
	}
}

public static void ToEMPBullet(IProjectile projectile)
{
	if (!m_customBullets.ContainsKey(projectile.InstanceID))
	{
		m_customBullets.Add(projectile.InstanceID, projectile);
	}
}

public static bool IsTouchingCircle(Area area, Vector2 center, float radius)
{
	var lines = new List<Vector2[]>()
	{
		new Vector2[] { area.BottomRight, area.BottomLeft },
		new Vector2[] { area.BottomLeft, area.TopLeft },
		new Vector2[] { area.TopLeft, area.TopRight },
		new Vector2[] { area.TopRight, area.BottomRight },
	};

	var minDistanceToCenter = float.MaxValue;

	foreach (var line in lines)
	{
		var distanceToCenter = FindDistanceToSegment(center, line[0], line[1]);
		if (distanceToCenter < minDistanceToCenter) minDistanceToCenter = distanceToCenter;
	}

	return minDistanceToCenter <= radius;
}

// https://stackoverflow.com/a/1501725/9449426
public static float FindDistanceToSegment(Vector2 point, Vector2 p1, Vector2 p2)
{
	// Return minimum distance between line segment vw and point point
	var lengthSquare = (float)(Math.Pow(p2.X - p1.X, 2) + Math.Pow(p2.Y - p1.Y, 2));  // i.e. |p2-p1|^2 -  avoid a sqrt
	if (lengthSquare == 0.0) return Vector2.Distance(point, p1);   // p1 == p2 case
	// Consider the line extending the segment, parameterized as p1 + t (p2 - p1).
	// We find projection of point point onto the line. 
	// It falls where t = [(point-p1) . (p2-p1)] / |p2-p1|^2
	// We clamp t from [0,1] to handle points outside the segment vw.
	var t = MathHelper.Clamp(Vector2.Dot(point - p1, p2 - p1) / lengthSquare, 0, 1);
	var projection = p1 + t * (p2 - p1);  // Projection falls on the segment
	return Vector2.Distance(point, projection);
}

public static List<int> DragonBreath = new List<int>();
public static List<int> DragonUpdated = new List<int>();

public static void dragonCreate(IProjectile p){
	
	if(Game.GetSingleObjectByCustomID("__DRAGON_BREATH_TIMER__") == null)

	if(m_dragonBreathUpdate == null)

	DragonUpdated.Add(p.InstanceID);
	Game.PlaySound("SawedOff", Vector2.Zero);
}

public static void dragonUpdate(float e)
{
	DragonBreath = DragonUpdated;
	DragonUpdated = new List<int>();
	
	foreach (int pID in DragonBreath)
	{
		IProjectile p = Game.GetProjectile(pID);
		
		if(p != null){
			
			if(!p.PowerupFireActive)
				Ignite(p);
				
			
			if (p.TotalDistanceTraveled > 300 && !p.IsRemoved)
			{
				p.FlagForRemoval();
				Game.SpawnFireNode(p.Position, p.Direction * 8);
			}		
			
			if(!p.IsRemoved)
				DragonUpdated.Add(pID);
		}
	}
}

public static void Ignite(IProjectile p)
{

	float multi = 3;
	float crit = 0.01f;
	switch (p.ProjectileItem)
	{
		case(ProjectileItem.REVOLVER):
			multi = 5;
			break;
		case(ProjectileItem.SNIPER):
		case(ProjectileItem.MAGNUM):
			multi = 8;
			break;
		case(ProjectileItem.SHOTGUN):
		case(ProjectileItem.DARK_SHOTGUN):
		case(ProjectileItem.SAWED_OFF):
			multi = 2f;
			break;
		case(ProjectileItem.GRENADE_LAUNCHER):
		case(ProjectileItem.BAZOOKA):
			multi = 15;
			crit = 0.2f;
			break;				
	}
	
	for (int i = 0; i < multi; i++)
	{
		var pro = Game.SpawnProjectile(ProjectileItem.SILENCEDUZI, p.Position, p.Direction + new Vector2(rnd.Next(-10, 10) / 100f, rnd.Next(-10, 10) / 100f));
		pro.DamageDealtModifier *= 0.1f;
		pro.CritChanceDealtModifier *= crit;
		DragonUpdated.Add(pro.InstanceID);
		pro.PowerupFireActive = true;
	}
	
	p.FlagForRemoval();
}

				Game.PlayEffect("S_P", p.Position);
			p.Direction = p.Direction + new Vector2(rnd.Next(-100, 100) / 700f, rnd.Next(-100, 100) / 700f);
		}
	}

public static void pop(TriggerArgs args){

	if(args.Caller is IObject){
		string oof = (args.Caller as IObject).CustomID;
				
		foreach(IObject obj in Game.GetObjectsByCustomID(oof)){
			Game.PlaySound("PlayerJump", new Vector2(0,0), 100);
			obj.Remove();
		}	
	}
}

public static void laserCreate(IProjectile p)
{
	//Create Laser
	var dir = p.Direction;
	dir.Normalize();
	var pos = p.Position;
	var pos2 = pos + dir * 500;
	IObject t = null;
	RayCastInput input = new RayCastInput(true) { ProjectileHit = RayCastFilterMode.True, AbsorbProjectile = RayCastFilterMode.True };
	RayCastResult[] rays = Game.RayCast(pos, pos2, input);
	if (rays.Length > 0)
	{
		if (rays[0].Hit)
		{
			pos2 = rays[0].Position;
			t = rays[0].HitObject;
		}
	}
	if (t != null)
	{
		t.SetMaxFire();
		t.SetLinearVelocity(t.GetLinearVelocity() + dir * 4);
		if (t is IPlayer)
		if (t is IPlayer)
			t.SetWorldPosition(t.GetWorldPosition() + new Vector2(0, 2));
		t.DealDamage(GetBulletDamage(p.ProjectileItem));
		Game.SpawnFireNode(pos2, dir);
	}
	//Draw trail
	int count = (int)Vector2.Distance(pos, pos2);
	for (int i = 0; i < count; i++)
	{
		var pp = pos + ((pos2 - pos) / count) * i;
		Game.PlayEffect("TR_B", pp);
		Game.PlayEffect("GLM", pp);
		if (p.ProjectileItem == ProjectileItem.BAZOOKA || p.ProjectileItem == ProjectileItem.GRENADE_LAUNCHER)
		{
			Game.PlayEffect("GLM", pp + new Vector2(0, 1));
			Game.PlayEffect("GLM", pp + new Vector2(0, -1));
		}
	}
	Game.PlayEffect("Electric", pos);
	p.FlagForRemoval();
}
